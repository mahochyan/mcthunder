    def build(self):

        self.box('hull', (0.42, 1.24, -2.17), (0.52, 0.10, 0.36), 'paint')

            self.rod('hull', (0.42, 1.31, -2.10 + dz), (0.42, 1.42, -2.10 + dz), 0.015, 'steel', n=5, cap_a=False)

        # headlights on brackets at hull front corners (round lamp + grid guard)
        for s in (-1, 1):

            self.rod('hull', (s * 0.88, 0.94, -2.96), (s * 0.88, 1.18, -2.96), 0.013, 'steel', n=4, cap_a=False, cap_b=False)
            self.rod('hull', (s * 0.88 - 0.11, 1.06, -2.96), (s * 0.88 + 0.11, 1.06, -2.96), 0.013, 'steel', n=4, cap_a=False, cap_b=False)
